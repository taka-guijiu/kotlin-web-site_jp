訳者注）関数の色々な記述方法を以下のとおり記す。

fun uppercaseString(text : String) 
	{ text.uppercase() }  // 標準的な関数の書き方(p.1)
     
fun main(args: Array<String>) {
    println(uppercaseString("hello world"))
}

//-------------------------------------------------------------------

fun uppercaseString(text : String) 
	= text.uppercase()   // 単一式による書き方(p.3)

fun main(args: Array<String>) {
    println(uppercaseString("hello world"))
}

//-------------------------------------------------------------------

fun main(args: Array<String>) {
    val upperCaseString 
    	= { text: String -> text.uppercase() }    // ラムダ式(p.8)
    
    println(upperCaseString("hello"))
}
//-------------------------------------------------------------------
